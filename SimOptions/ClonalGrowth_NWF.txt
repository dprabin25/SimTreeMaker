mutationIntroTick            = 1                    # Generation when driver mutation (m2) is injected (1 = from the very start)
mutationPosition             = 5000                 # Chromosome position where m2 is placed (0 – chromosomeEnd)

treeOutputFile               = ClonalGrowth_NWF    # Base name for output files (tree, newick, PNG folders)

min_samples_per_generation   = 1              # Min cells per generation — generations with fewer cells are skipped
